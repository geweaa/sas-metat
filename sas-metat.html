  inner_join(tax, by = "orf") %>%
  group_by(phylum, sample) %>%
  # Sum the abundance of each phylum within a sample
  summarise(tpm = sum(tpm), .groups = 'drop_last') %>%
  # Calculate the mean abundance of each phylum over the categories
  summarise(mean.tpm = sum(tpm), .groups = 'drop') %>%

nmds <- tpm %>%
  filter(!sample %in% c("S1.1", "S1.2", "S2.1", "S2.2")) %>%
  # Sum the abundance of both coverage per metagenome
  inner_join(eggnog, by = "orf") %>%
  # Filter out NA and add multiple KOs on new line
  filter(kegg_module != "-") %>% separate_rows(kegg_module, sep = ",") %>%
  # Sum the abundance of each KO within the metagenomes
  group_by(sample, kegg_module) %>% 
  summarise(tpm = sum(tpm), .groups = "drop") %>%
  # Pivot to prepare a numerical matrix
  pivot_wider(names_from = "kegg_module", values_from = "tpm", values_fill = 0) %>%
  column_to_rownames("sample") %>%
  vegan::metaMDS()
